import socket,sys,threading
port=int(sys.argv[1]); n=int(sys.argv[2])
srv=socket.socket(); srv.setsockopt(socket.SOL_SOCKET,socket.SO_REUSEADDR,1)
srv.bind(("127.0.0.1",port)); srv.listen(20)
def h(c):
    d=b""
    while b"\r\n\r\n" not in d: d+=c.recv(8192)
    c.sendall(b"HTTP/1.1 200 OK\r\nContent-Length: "+str(n).encode()+b"\r\n\r\n")
    blk=b"A"*65536
    sent=0
    while sent<n:
        k=min(len(blk),n-sent); c.sendall(blk[:k]); sent+=k
    c.close()
while True:
    c,_=srv.accept(); threading.Thread(target=h,args=(c,),daemon=True).start()

import socketserver,sys
MODE=sys.argv[1]
class H(socketserver.BaseRequestHandler):
    def handle(self):
        d=b""
        while b"\r\n\r\n" not in d:
            c=self.request.recv(4096)
            if not c: return
            d+=c
        if MODE=="chunked":
            self.request.sendall(b"HTTP/1.1 200 OK\r\nTransfer-Encoding: chunked\r\n\r\n5\r\nHELLO\r\n6\r\n WORLD\r\n0\r\n\r\n")
        elif MODE=="loc":
            self.request.sendall(b"HTTP/1.1 302 Found\r\nLocation: /upstream-target\r\nContent-Length: 2\r\n\r\nhi")
        elif MODE=="nocl":
            self.request.sendall(b"HTTP/1.1 200 OK\r\n\r\nno-cl-body")
        elif MODE=="status999":
            self.request.sendall(b"HTTP/1.1 999 Weird\r\nContent-Length: 1\r\n\r\nZ")
class S(socketserver.ThreadingTCPServer):
